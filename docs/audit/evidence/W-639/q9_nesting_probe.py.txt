DB = TMP / "probe.db"
con = sqlite3.connect(DB, isolation_level=None)
con.execute("CREATE TABLE rows (id TEXT PRIMARY KEY, who TEXT)")


def insert(row_id: str, who: str) -> dict:
    con.execute("INSERT INTO rows VALUES (?, ?)", (row_id, who))
    return {"id": row_id, "who": who}


def delete(row_id: str) -> None:
    con.execute("DELETE FROM rows WHERE id = ?", (row_id,))


def rows() -> list[tuple]:
    return [tuple(r) for r in con.execute("SELECT id, who FROM rows ORDER BY id")]


def new_authorizer(tag: str, actor: str = "EXECUTIVE", shared_dir: Path | None = None):
    d = shared_dir or (TMP / tag)
    d.mkdir(parents=True, exist_ok=True)
    return ConstitutionalAuthorizer(actor=actor, idempotency_ledger_path=d / "idem.json")


def ledger_dump(auth) -> list[dict]:
    out = []
    led = auth.boundary.idempotency_ledger
    for r in led.all_records():
        out.append(
            {
                "key": r.key.composite,
                "status": r.status.value,
                "applied": list(r.applied_effect_signatures),
                "attempts": len(r.attempts),
                "failure": (r.failure_reason or "")[:110],
            }
        )
    return out


def comp_dump(auth) -> list[dict]:
    j = auth.boundary.compensation_journal
    return [
        {
            "id": c.compensation_id,
            "status": getattr(c.status, "value", str(c.status)),
            "residual": list(c.residual_signatures),
        }
        for c in j.all_records()
    ]


def gw_dump(auth) -> list[dict]:
    return [
        {"action": r.action, "actor": r.actor, "decision": str(r.decision), "enforcement": r.enforcement, "fp": r.fingerprint[:12]}
        for r in auth.gateway.records
    ]


def show(title: str, auth, extra: dict | None = None, other=None) -> None:
    print(f"\n=== {title} ===")
    print("rows(DB):", rows())
    print("idempotency(ledger of outer authorizer):", json.dumps(ledger_dump(auth), ensure_ascii=False, indent=1))
    if other is not None and other is not auth:
        print("idempotency(ledger of inner authorizer instance):", json.dumps(ledger_dump(other), ensure_ascii=False, indent=1))
    print("compensation journal:", comp_dump(auth))
    print("gateway records (outer authorizer):", gw_dump(auth))
    if other is not None and other is not auth:
        print("gateway records (inner authorizer):", gw_dump(other))
    print("outbox configured on boundary:", auth.boundary.outbox is not None)
    if extra:
        for k, v in extra.items():
            print(f"{k}: {v}")


def exc_chain(exc: BaseException) -> str:
    parts = []
    e: BaseException | None = exc
    while e is not None:
        parts.append(f"{type(e).__name__}: {str(e)[:140]}")
        e = e.__cause__ or e.__context__
    return " <- ".join(parts)


def inner_call(auth, tag: str, *, fail: bool = False, key_value: str | None = None,
               action: str = "q9.inner", target: str | None = None):
    target = target or f"probe/{tag}/inner"
    eff = declared_effect("CREATE", target, "inner row")

    def _apply(_e):
        r = insert(f"{tag}-inner", "inner")
        if fail:
            raise RuntimeError("inner applier fails AFTER writing its row")
        return r

    return auth.guard_declared(
        action,
        target,
        declared_effects=(eff,),
        applier=_apply,
        operation_key=operation_key("q9.inner", key_value or f"{tag}"),
        compensators=(compensator(eff.signature, lambda: delete(f"{tag}-inner"), "delete inner"),),
    )


def scenario(name, fn):
    con.execute("DELETE FROM rows")
    print("\n" + "#" * 78 + f"\n# {name}\n" + "#" * 78)
    try:
        fn()
    except Exception as exc:  # noqa: BLE001 — المسبارُ يطبعُ ولا يبتلعُ صامتًا
        print("UNEXPECTED (probe-level):", exc_chain(exc))
        traceback.print_exc(limit=3)


# ── S1: تعشيشٌ ناجحٌ على مُصرِّحٍ مشترك ─────────────────────────────────────
def s1():
    auth = new_authorizer("s1")
    tag = "s1"
    eff = declared_effect("CREATE", f"probe/{tag}/outer", "outer row")
    inner_results = []

    def _apply(_e):
        inner_results.append(inner_call(auth, tag))
        return insert(f"{tag}-outer", "outer")

    res = auth.guard_declared(
        "q9.outer", f"probe/{tag}/outer", declared_effects=(eff,), applier=_apply,
        operation_key=operation_key("q9.outer", tag),
        compensators=(compensator(eff.signature, lambda: delete(f"{tag}-outer"), "delete outer"),),
    )
    inner = inner_results[0]
    recs = auth.gateway.records
    show("S1 nested success (shared authorizer)", auth, {
        "outer.permit_id": res.outcome.permit_id,
        "inner.permit_id": inner.outcome.permit_id,
        "outer.outcome.applied_effects": [e.signature for e in res.outcome.applied_effects],
        "outer.evidence.request_fingerprint": res.evidence.request_fingerprint[:12],
        "gateway fp of OUTER decision (first q9.outer record)": next(r for r in recs if r.action == "q9.outer").fingerprint[:12],
        "gateway fp of INNER decision (first q9.inner record)": next(r for r in recs if r.action == "q9.inner").fingerprint[:12],
        "gateway records count (outer+inner on one gateway)": len(recs),
        "outer evidence is INNER's record?": res.evidence.request_fingerprint == recs[-1].fingerprint
        and recs[-1].action == "q9.inner",
    })


# ── S2: الخارجيُّ يفشلُ بعدَ التزامِ الداخليّ (أثرٌ ثانٍ للخارجيِّ يفشل) ───────────
def s2():
    auth = new_authorizer("s2")
    tag = "s2"
    e1 = declared_effect("CREATE", f"probe/{tag}/outer/a", "outer a (calls inner)")
    e2 = declared_effect("CREATE", f"probe/{tag}/outer/b", "outer b (fails)")

    def _apply(e):
        if e.signature == e1.signature:
            inner_call(auth, tag)
            return insert(f"{tag}-outer-a", "outer")
        raise RuntimeError("outer second effect fails after inner committed")

    try:
        auth.guard_declared(
            "q9.outer", f"probe/{tag}/outer", declared_effects=(e1, e2), applier=_apply,
            operation_key=operation_key("q9.outer", tag),
            compensators=(
                compensator(e1.signature, lambda: delete(f"{tag}-outer-a"), "delete outer a ONLY"),
                compensator(e2.signature, lambda: delete(f"{tag}-outer-b"), "delete outer b"),
            ),
        )
    except Exception as exc:  # noqa: BLE001
        print("raised to caller:", exc_chain(exc))
    show("S2 outer fails after inner committed — outer compensator undoes only its own row", auth)


# ── S2b: معوِّضُ الخارجيِّ يعكسُ أثرَ الداخليِّ (نمطُ open_case) ثمَّ إعادةٌ بالمفاتيحِ نفسِها ──
def s2b():
    auth = new_authorizer("s2b")
    tag = "s2b"
    e1 = declared_effect("CREATE", f"probe/{tag}/outer/a", "outer a (calls inner)")
    e2 = declared_effect("CREATE", f"probe/{tag}/outer/b", "outer b")
    state = {"fail_b": True}

    def _apply(e):
        if e.signature == e1.signature:
            inner = inner_call(auth, tag)  # مفتاحُ الداخليِّ حتميٌّ (مشتقٌّ من المرجع)
            print("  inner.is_replay =", inner.is_replay)
            return insert(f"{tag}-outer-a", "outer")
        if state["fail_b"]:
            raise RuntimeError("outer b fails (attempt 1)")
        return insert(f"{tag}-outer-b", "outer")

    def comp_a():
        delete(f"{tag}-outer-a")
        delete(f"{tag}-inner")  # يعكسُ أثرَ الداخليِّ مباشرةً — بلا علمِ سجلِّ الداخليّ

    kw = dict(
        declared_effects=(e1, e2), applier=_apply,
        operation_key=operation_key("q9.outer", tag),
        compensators=(
            compensator(e1.signature, comp_a, "delete outer a + inner row"),
            compensator(e2.signature, lambda: delete(f"{tag}-outer-b"), "delete outer b"),
        ),
    )
    try:
        auth.guard_declared("q9.outer", f"probe/{tag}/outer", **kw)
    except Exception as exc:  # noqa: BLE001
        print("attempt 1 raised:", exc_chain(exc))
    show("S2b after attempt 1 (outer compensated inner's row; inner ledger?)", auth)
    state["fail_b"] = False
    time.sleep(1.1)  # Q-11: هويّةُ الإذنِ بدقّةِ ثانية — نتجنّبُ تصادمَه ليُقاسَ التعشيشُ وحدَه
    try:
        res = auth.guard_declared("q9.outer", f"probe/{tag}/outer", **kw)
        print("attempt 2 outcome: is_replay =", res.is_replay)
    except Exception as exc:  # noqa: BLE001
        print("attempt 2 raised:", exc_chain(exc))
    show("S2b after attempt 2 (retry with same keys)", auth, {
        "inner row present?": any(r[0] == f"{tag}-inner" for r in rows()),
    })

# ── S4: الداخليُّ يفشلُ (بعدَ أثرٍ سابقٍ للخارجيّ) ─────────────────────────────
def s4():

            compensators=(

            ),
        )
    except Exception as exc:  # noqa: BLE001
        print("raised to caller:", exc_chain(exc))

# ── S4b: الداخليُّ يفشلُ والخارجيُّ يبتلعُ الاستثناءَ داخلَ مُطبِّقِه ─────────────
def s4b():

    res = auth.guard_declared(

    show("S4b inner fails, outer swallows — outer SUCCEEDED?", auth, {"outer.is_replay": res.is_replay})


# ── S5: تعشيشٌ بالمفتاحِ نفسِه (إعادةُ دخولٍ حرفيّة) ───────────────────────────
def s5():
    auth = new_authorizer("s5")
    tag = "s5"
    eff = declared_effect("CREATE", f"probe/{tag}/outer", "outer row")
    key = operation_key("q9.same", tag)

    def _apply(_e):
        eff_i = declared_effect("CREATE", f"probe/{tag}/inner", "inner")
        auth.guard_declared(
            "q9.inner", f"probe/{tag}/inner", declared_effects=(eff_i,),
            applier=lambda _x: insert(f"{tag}-inner", "inner"), operation_key=key,
            compensators=(compensator(eff_i.signature, lambda: delete(f"{tag}-inner")),),
        )
        return insert(f"{tag}-outer", "outer")

    try:
        auth.guard_declared(
            "q9.outer", f"probe/{tag}/outer", declared_effects=(eff,), applier=_apply,
            operation_key=key,
            compensators=(compensator(eff.signature, lambda: delete(f"{tag}-outer"), "delete outer"),),
        )
    except Exception as exc:  # noqa: BLE001
        print("raised to caller:", exc_chain(exc))
    show("S5 same operation_key nested (literal re-entry)", auth)


# ── S6: مُصرِّحانِ منفصلانِ (كلُّ خدمةٍ تبني مُصرِّحَها) على سجلِّ ذرّيّةٍ واحد ──────

# ── S7: فعلُ مالٍ متعشِّشٌ بفاعلِ الخزانة (هل تأذنُ البوابةُ للتعشيشِ بين فاعلَين؟) ──
def s7():
    exec_auth = new_authorizer("s7e")
    treas_auth = new_authorizer("s7t", actor="TREASURY")
    tag = "s7"
    eff = declared_effect("WRITE", f"probe/{tag}/expenditure", "wrapper status")

    def _apply(_e):
        r = inner_call(treas_auth, tag, action="disburse_funds", target=f"probe/{tag}/treasury/txn")
        print("  inner disburse_funds decision:", gw_dump(treas_auth))
        return insert(f"{tag}-outer", "outer") | {"inner_permit": r.outcome.permit_id}

    try:
        res = exec_auth.guard_declared(
            "execute_transfer", f"probe/{tag}/expenditure", declared_effects=(eff,), applier=_apply,
            operation_key=operation_key("q9.outer", tag),
            compensators=(compensator(eff.signature, lambda: delete(f"{tag}-outer"), "revert outer"),),
        )
        print("outer value:", res.value)
    except Exception as exc:  # noqa: BLE001
        print("raised to caller:", exc_chain(exc))
    show("S7 money verbs: EXECUTIVE execute_transfer -> TREASURY disburse_funds", exec_auth, other=treas_auth)
