def run_pytest(k: str) -> int:
    env = dict(**{k: v for k, v in __import__("os").environ.items()})
    env["AMOS_DATABASE_URL"] = DB
    proc = subprocess.run(
        [str(PY), "-m", "pytest", TEST, "-q", "--tb=line", "-k", k, "-p", "no:cacheprovider"],
        cwd=REPO,
        env=env,
        text=True,
    )
    return proc.returncode


def apply(path: Path, old: str, new: str, label: str) -> None:
    text = path.read_text(encoding="utf-8")
    count = text.count(old)
    if count != 1:
        raise SystemExit(f"{label}: عددُ التطابقاتِ {count} لا 1")
    path.write_text(text.replace(old, new, 1), encoding="utf-8")


def restore() -> None:
    subprocess.run(
        ["git", "checkout", "--", str(RULINGS.relative_to(REPO)), str(BOUNDARY.relative_to(REPO))],
        cwd=REPO,
        check=True,
    )
